// 1. SOURCE ATOM
const query = atom('');

// 2. ASYNC COMPUTED PIPE
const data = computed(async () => {
  if (!query.value) return [];
  return await api(query.value);
}, { defaultValue: [] });

// 3. REACTIVE SYNC
effect(() => {
  if (data.isPending) load();
  else render(data.value);
});
